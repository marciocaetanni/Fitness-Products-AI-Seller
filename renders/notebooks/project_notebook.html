            print(f"❌ The {file} is corrupted!")

# Function with restarting download
def download_with_resume(url, file_path, chunk_size=8192):
    # Verifying if a partial file exists
    temp_size = os.path.getsize(file_path) if os.path.exists(file_path) else 0

    # Verifying if it is accepted range download
    head = requests.head(url)
    accept_ranges = head.headers.get('Accept-Ranges', 'none')
    total_size = int(head.headers.get('Content-Length', 0))

    if accept_ranges != 'bytes':
        temp_size = 0  

    headers = {'Range': f"bytes={temp_size}-"} if temp_size > 0 else {}

    # Stream download
    with requests.get(url, headers=headers, stream=True) as r:
        r.raise_for_status()
        mode = 'ab' if temp_size > 0 else 'wb'
        with open(file_path, mode) as f:
            progress = tqdm(
                total=total_size,
                initial=temp_size,
                unit='B',
                unit_scale=True,
                unit_divisor=1024,
                desc=os.path.basename(file_path),
                ncols=80
            )
            for chunk in r.iter_content(chunk_size=chunk_size):
                if chunk:
                    f.write(chunk)
                    progress.update(len(chunk))
            progress.close()
    print(f"✔️ {os.path.basename(file_path)} finished!")
